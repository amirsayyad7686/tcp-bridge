		HAL_GPIO_WritePin(FLASHER_GPIO_Port,FLASHER_Pin,1);

		HAL_GPIO_WritePin(HORN_GPIO_Port,HORN_Pin,0);
		HAL_GPIO_WritePin(FLASHER_GPIO_Port,FLASHER_Pin,0);
		HAL_Delay(800);
	}
}

		HAL_Delay(80);

		HAL_GPIO_TogglePin(FLASHER_GPIO_Port,FLASHER_Pin);
		HAL_Delay(80);
		HAL_GPIO_TogglePin(HORN_GPIO_Port,HORN_Pin);
		HAL_GPIO_TogglePin(FLASHER_GPIO_Port,FLASHER_Pin);

		HAL_GPIO_TogglePin(FLASHER_GPIO_Port,FLASHER_Pin);
		HAL_Delay(80);

void OperationOn(){
	MX_TIM3_Init();
	x = 0;
	HAL_TIM_Base_Start_IT(&htim3);
	KeylessStep = 0;
	openIT_flag = false;
	Keyless = false;

	while(Operation && !Close){
		if(openIT_flag){
			//check
			openIT_flag = false;
			HAL_GPIO_TogglePin(HORN_GPIO_Port,HORN_Pin);
			HAL_GPIO_TogglePin(FLASHER_GPIO_Port,FLASHER_Pin);
			HAL_Delay(80);
			HAL_GPIO_TogglePin(HORN_GPIO_Port,HORN_Pin);
			HAL_GPIO_TogglePin(FLASHER_GPIO_Port,FLASHER_Pin);
			HAL_Delay(80);
			HAL_GPIO_TogglePin(HORN_GPIO_Port,HORN_Pin);
			HAL_GPIO_TogglePin(FLASHER_GPIO_Port,FLASHER_Pin);
			HAL_Delay(80);
			HAL_GPIO_WritePin(HORN_GPIO_Port,HORN_Pin,0);
			HAL_GPIO_WritePin(FLASHER_GPIO_Port,FLASHER_Pin,0);
		}
		if(HAL_GPIO_ReadPin(Keyless_GPIO_Port,Keyless_Pin) == 1){
			KeylessStep = KeylessStep + 1;
			HAL_TIM_Base_Stop_IT(&htim3);
			if(HAL_GPIO_ReadPin(Break_GPIO_Port,Break_Pin) == 1){
				HAL_GPIO_WritePin(ACC_GPIO_Port,ACC_Pin,1);
				HAL_Delay(1000);
				HAL_GPIO_WritePin(IGN_GPIO_Port,IGN_Pin,1);
				HAL_Delay(1000);
				HAL_GPIO_WritePin(START_GPIO_Port,START_Pin,1);
				HAL_GPIO_WritePin(IGN_GPIO_Port,IGN_Pin,0);
				HAL_GPIO_WritePin(ACC_GPIO_Port,ACC_Pin,0);
				HAL_Delay(1000);
				HAL_GPIO_WritePin(IGN_GPIO_Port,START_Pin,0);
				HAL_GPIO_WritePin(IGN_GPIO_Port,IGN_Pin,1);
				HAL_GPIO_WritePin(ACC_GPIO_Port,ACC_Pin,1);
				KeylessStep = 0;
				//return;
			}
			if(HAL_GPIO_ReadPin(Keyless_GPIO_Port,Keyless_Pin) == 1 && KeylessStep <= 1){
				HAL_GPIO_WritePin(ACC_GPIO_Port,ACC_Pin,1);
				HAL_Delay(1000);
				
			}
			if(HAL_GPIO_ReadPin(Keyless_GPIO_Port,Keyless_Pin) == 1 && KeylessStep <= 2){
				KeylessStep = 0;
				HAL_GPIO_WritePin(IGN_GPIO_Port,IGN_Pin,1);
				HAL_Delay(1000);
					if(HAL_GPIO_ReadPin(Keyless_GPIO_Port,Keyless_Pin) == 1){
						//KeylessStep = 0;
						CarOff();
						HAL_Delay(1000);
					}
					
			}
					
		}
	}
}
void Alarm(){
	y = 0;
	CarOff();
	MX_TIM2_Init();
	HAL_TIM_Base_Start_IT(&htim2);
	if(!timer2_flag){
		timer2_flag = true;
	}
	while (Close){
		SercurityOn();
	}
}

}
